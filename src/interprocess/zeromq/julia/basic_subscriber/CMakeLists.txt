set(JULIA_SUB_OUT_DIR "${project_BUILD_DIR}/julia/basic_subscriber")
file(MAKE_DIRECTORY "${JULIA_SUB_OUT_DIR}")

# Copy the Julia script to the build output directory so it sits alongside
# the generated .so and can use relative paths to find protobuf/protobuf.jl
# and Goby.jl.
configure_file(subscriber.jl "${JULIA_SUB_OUT_DIR}/subscriber.jl" COPYONLY)

# Build the C++ shared library that bridges BasicJuliaSubscriber to Julia.
goby_generate_julia(
    basic_julia_subscriber
    "${JULIA_SUB_OUT_DIR}"
    "${CMAKE_CURRENT_SOURCE_DIR}/interface.yml"
    "${CMAKE_CURRENT_SOURCE_DIR}/config.proto"
    messages/nav.pb.h
    messages/groups.h
)

# nav.pb.h is generated by the goby3_example_messages target.
add_dependencies(basic_julia_subscriber goby3_example_messages)
target_link_libraries(basic_julia_subscriber goby3_example_messages)

# Depend on the Julia protobuf generation for nav.proto.
add_dependencies(basic_julia_subscriber julia_proto_nav)
