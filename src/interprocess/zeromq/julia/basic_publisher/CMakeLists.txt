set(JULIA_PUB_OUT_DIR "${project_BUILD_DIR}/julia/basic_publisher")
file(MAKE_DIRECTORY "${JULIA_PUB_OUT_DIR}")

# Copy the Julia script to the build output directory so it sits alongside
# the generated .so and can use relative paths to find protobuf/protobuf.jl
# and Goby.jl.
configure_file(publisher.jl "${JULIA_PUB_OUT_DIR}/publisher.jl" COPYONLY)

# Build the C++ shared library that bridges BasicJuliaPublisher to Julia.
# The extra arguments are header file paths (relative to project_INC_DIR) that
# gen_goby.jl will emit as #include directives in the generated C++ source.
goby_generate_julia(
    basic_julia_publisher
    "${JULIA_PUB_OUT_DIR}"
    "${CMAKE_CURRENT_SOURCE_DIR}/interface.yml"
    "${CMAKE_CURRENT_SOURCE_DIR}/config.proto"
    messages/nav.pb.h
    messages/groups.h
)

# nav.pb.h is generated by the goby3_example_messages target, so we must
# build that target before trying to compile the C++ wrapper.
add_dependencies(basic_julia_publisher goby3_example_messages)
target_link_libraries(basic_julia_publisher goby3_example_messages)

# Depend on the Julia protobuf generation for nav.proto.
add_dependencies(basic_julia_publisher julia_proto_nav)
